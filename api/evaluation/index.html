    def detect_r_peaks(
    signal: ArrayLike,
    fs: int,
    lead: Optional[int] = None,
    method: str = "simple",
) -> np.ndarray:
    """
    R-peak sample indices of one ECG ``[C, T]`` (or ``[T]``).

    Args:
        signal: One ECG.
        fs: Sampling rate in Hz.
        lead: Lead to use; default = the lead with the largest QRS energy.
        method: ``"simple"`` (band-pass + peak picking, no extra dependency) or
            ``"neurokit"`` (``neurokit2.ecg_peaks``; needs ``ecgen[eval]``).
    """
    x = _as_numpy(signal)
    if x.ndim == 1:
        x = x[None]
    if x.shape[-1] < int(0.5 * fs):
        return np.array([], dtype=int)

    nyq = fs / 2
    b, a = butter(2, [5 / nyq, min(30 / nyq, 0.99)], btype="band")
    filtered = filtfilt(b, a, x, axis=-1)
    if lead is None:
        lead = int(np.argmax(np.percentile(np.abs(filtered), 99, axis=-1)))
    sig = filtered[lead]

    if method == "neurokit":
        import neurokit2 as nk

        _, info = nk.ecg_peaks(x[lead], sampling_rate=fs)
        return np.asarray(info["ECG_R_Peaks"], dtype=int)
    if method != "simple":
        raise ValueError(f"Unknown method {method!r}")

    # QRS polarity varies by lead: use the dominant sign
    if np.abs(sig.min()) > np.abs(sig.max()):
        sig = -sig
    height = 0.4 * np.percentile(sig, 99.5)
    peaks, _ = find_peaks(sig, height=height, distance=int(0.25 * fs))
    return peaks

    def heart_rate(
    signals: ArrayLike,
    fs: int,
    method: str = "simple",
    lead: Optional[int] = None,
) -> np.ndarray:
    """
    Heart rate (bpm) per ECG from the median RR interval; NaN when fewer than
    two beats are found.

    Args:
        signals: ``[N, C, T]``, ``[C, T]``, or ``[T]``.
        fs: Sampling rate in Hz.
    """
    x = _as_numpy(signals)
    if x.ndim == 1:
        x = x[None, None]
    elif x.ndim == 2:
        x = x[None]
    rates: List[float] = []
    for ecg in x:
        peaks = detect_r_peaks(ecg, fs, lead=lead, method=method)
        if len(peaks) < 2:
            rates.append(np.nan)
        else:
            rates.append(60.0 * fs / float(np.median(np.diff(peaks))))
    return np.asarray(rates)

    def adherence(requested: Sequence[float], measured: Sequence[float]) -> dict:
    """
    How well measured values follow requested ones (NaNs ignored).

    Returns ``{"mae", "pearson_r", "n_valid", "n"}``.
    """
    req = np.asarray(requested, dtype=np.float64)
    mea = np.asarray(measured, dtype=np.float64)
    ok = ~(np.isnan(req) | np.isnan(mea))
    out = {
        "n": int(len(req)),
        "n_valid": int(ok.sum()),
        "mae": float("nan"),
        "pearson_r": float("nan"),
    }
    if ok.sum() >= 2:
        out["mae"] = float(np.mean(np.abs(req[ok] - mea[ok])))
        if np.std(req[ok]) > 0 and np.std(mea[ok]) > 0:
            out["pearson_r"] = float(np.corrcoef(req[ok], mea[ok])[0, 1])
    return out

    def handcrafted_features(x: ArrayLike, fs: int = 500) -> np.ndarray:
    """
    Per-ECG feature vector ``[N, D]``:

    - heart rate (bpm, NaN → batch median);
    - per lead: standard deviation, 1st/99th percentile, and the relative power
      in the bands 0.5–5, 5–15, 15–40, 40–100 Hz;
    - the upper triangle of the inter-lead correlation matrix.
    """
    x = _as_numpy(x)
    if x.ndim == 2:
        x = x[None]
    n, c, t = x.shape
    hr = heart_rate(x, fs)
    hr = np.where(np.isnan(hr), np.nanmedian(hr) if np.isfinite(hr).any() else 0.0, hr)

    std = x.std(axis=-1)
    p1, p99 = np.percentile(x, 1, axis=-1), np.percentile(x, 99, axis=-1)
    freqs, psd = welch(x, fs=fs, nperseg=min(t, 2 * fs), axis=-1)
    total = psd.sum(-1, keepdims=True) + 1e-12
    bands = (
        np.stack(
            [
                psd[..., (freqs >= lo) & (freqs < min(hi, fs / 2))].sum(-1)
                for lo, hi in BANDS
            ],
            axis=-1,
        )
        / total
    )
    iu = np.triu_indices(c, k=1)
    corr = (
        np.stack([np.nan_to_num(np.corrcoef(e))[iu] for e in x])
        if c > 1
        else np.zeros((n, 0))
    )
    feats = [hr[:, None], std, p1, p99, bands.reshape(n, -1), corr]
    return np.concatenate(feats, axis=1).astype(np.float64)

    def module_features(
    model: torch.nn.Module, batch_size: int = 64, device: Optional[str] = None
) -> Extractor:
    """Wrap a ``[B, C, T] -> [B, D]`` network as an extractor."""
    device = device or ("cuda" if torch.cuda.is_available() else "cpu")

    def extract(x: ArrayLike) -> np.ndarray:
        xt = torch.as_tensor(_as_numpy(x), dtype=torch.float32)
        model.eval().to(device)
        out = []
        with torch.no_grad():
            for i in range(0, len(xt), batch_size):
                out.append(
                    model(xt[i : i + batch_size].to(device)).float().cpu().numpy()
                )
        return np.concatenate(out)

    return extract

    def frechet_distance(
    feats_a: np.ndarray, feats_b: np.ndarray, eps: float = 1e-6
) -> float:
    """Fréchet distance between Gaussians fitted to two feature sets (FID-style)."""
    mu_a, mu_b = feats_a.mean(0), feats_b.mean(0)
    sa, sb = (
        np.atleast_2d(np.cov(feats_a, rowvar=False)),
        np.atleast_2d(np.cov(feats_b, rowvar=False)),
    )
    covmean, _ = linalg.sqrtm(sa @ sb, disp=False)
    if not np.isfinite(covmean).all():
        off = np.eye(sa.shape[0]) * eps
        covmean = linalg.sqrtm((sa + off) @ (sb + off))
    covmean = covmean.real
    return float(
        ((mu_a - mu_b) ** 2).sum() + np.trace(sa) + np.trace(sb) - 2 * np.trace(covmean)
    )

    def mmd_rbf(
    feats_a: np.ndarray, feats_b: np.ndarray, bandwidth: Optional[float] = None
) -> float:
    """Unbiased squared MMD with an RBF kernel (bandwidth = median heuristic)."""
    x, y = np.asarray(feats_a, float), np.asarray(feats_b, float)
    z = np.concatenate([x, y])
    d2 = ((z[:, None] - z[None]) ** 2).sum(-1)
    if bandwidth is None:
        bandwidth = np.sqrt(0.5 * np.median(d2[d2 > 0])) if (d2 > 0).any() else 1.0
    k = np.exp(-d2 / (2 * bandwidth**2))
    n, m = len(x), len(y)
    kxx, kyy, kxy = k[:n, :n], k[n:, n:], k[:n, n:]
    return float(
        (kxx.sum() - np.trace(kxx)) / (n * (n - 1))
        + (kyy.sum() - np.trace(kyy)) / (m * (m - 1))
        - 2 * kxy.mean()
    )

    def precision_recall(
    real: np.ndarray, fake: np.ndarray, k: int = 3
) -> Dict[str, float]:
    """
    k-NN manifold precision / recall (Kynkäänniemi et al., 2019): precision = share
    of generated samples inside the real manifold, recall = share of real samples
    inside the generated manifold.
    """

    def radii(a: np.ndarray) -> np.ndarray:
        d = np.sqrt(((a[:, None] - a[None]) ** 2).sum(-1))
        return np.sort(d, axis=1)[:, min(k, len(a) - 1)]

    def inside(points: np.ndarray, ref: np.ndarray, r: np.ndarray) -> float:
        d = np.sqrt(((points[:, None] - ref[None]) ** 2).sum(-1))
        return float((d <= r[None]).any(axis=1).mean())

    return {
        "precision": inside(fake, real, radii(real)),
        "recall": inside(real, fake, radii(fake)),
    }

    def inter_lead_correlation_error(real: ArrayLike, fake: ArrayLike) -> float:
    """Mean |corr_real − corr_fake| over lead pairs, using mean correlation matrices (MIDT-ECG)."""

    def mean_corr(x: np.ndarray) -> np.ndarray:
        return np.nanmean(np.stack([np.corrcoef(e) for e in x]), axis=0)

    r, f = mean_corr(_as_numpy(real)), mean_corr(_as_numpy(fake))
    iu = np.triu_indices(r.shape[0], k=1)
    return float(np.abs(r[iu] - f[iu]).mean())

    def einthoven_residual(x: ArrayLike, lead_names: Sequence[str]) -> Dict[str, float]:
    """
    Relative residuals of the limb-lead identities (III = II − I,
    aVR = −(I + II)/2, aVL = I − II/2, aVF = II − I/2) after per-lead centering;
    ~0 for real recordings, larger when a model generates all 12 leads independently.
    """
    x = _as_numpy(x)
    x = x - x.mean(-1, keepdims=True)
    idx = {n: i for i, n in enumerate(lead_names)}
    need = ("I", "II", "III", "aVR", "aVL", "aVF")
    if not all(n in idx for n in need):
        raise ValueError(
            f"einthoven_residual needs leads {need}; got {list(lead_names)}"
        )
    i, ii = x[:, idx["I"]], x[:, idx["II"]]
    expected = {
        "III": ii - i,
        "aVR": -(i + ii) / 2,
        "aVL": i - ii / 2,
        "aVF": ii - i / 2,
    }
    out = {}
    for name, exp in expected.items():
        actual = x[:, idx[name]]
        out[name] = float(
            np.linalg.norm(actual - exp) / (np.linalg.norm(actual) + 1e-12)
        )
    out["mean"] = float(np.mean(list(out.values())))
    return out

    def reconstruction_metrics(
    pred: ArrayLike, true: ArrayLike, unobserved: ArrayLike
) -> Dict[str, float]:
    """
    MAE, RMSE and mean Pearson r on the unobserved entries (``unobserved``:
    ``[N, C, T]`` or ``[C]`` / ``[N, C]`` bool, True = to evaluate).
    """
    p, t = _as_numpy(pred), _as_numpy(true)
    m = np.asarray(_as_numpy(unobserved), dtype=bool)
    if m.ndim == 1:
        m = np.broadcast_to(m[None, :, None], p.shape)
    elif m.ndim == 2:
        m = np.broadcast_to(m[:, :, None], p.shape)
    err = (p - t)[m]
    rs = []
    for n in range(p.shape[0]):
        for c in range(p.shape[1]):
            sel = m[n, c]
            if sel.sum() > 2 and p[n, c][sel].std() > 0 and t[n, c][sel].std() > 0:
                rs.append(np.corrcoef(p[n, c][sel], t[n, c][sel])[0, 1])
    return {
        "mae": float(np.abs(err).mean()) if err.size else float("nan"),
        "rmse": float(np.sqrt((err**2).mean())) if err.size else float("nan"),
        "pearson_r": float(np.mean(rs)) if rs else float("nan"),
    }

    query: ArrayLike, reference: ArrayLike, batch: int = 256
) -> np.ndarray:
    """L2 distance from each query ECG to its nearest reference ECG (flattened signals)."""
    q = _as_numpy(query).reshape(len(_as_numpy(query)), -1)
    r = _as_numpy(reference).reshape(len(_as_numpy(reference)), -1)
    r2 = (r**2).sum(1)
    out = []
    for i in range(0, len(q), batch):
        qb = q[i : i + batch]
        d2 = (qb**2).sum(1)[:, None] + r2[None] - 2 * qb @ r.T
        out.append(np.sqrt(np.clip(d2.min(1), 0, None)))
    return np.concatenate(out)

    class ResNet1dClassifier(nn.Module):
    """``[B, C, T] -> [B, num_classes]`` logits; :meth:`features` gives the pooled embedding."""

    def __init__(
        self,
        in_channels: int = 12,
        num_classes: int = 5,
        width: int = 64,
        depth: int = 4,
    ) -> None:
        super().__init__()
        self.stem = nn.Sequential(
            nn.Conv1d(in_channels, width, 15, 2, 7, bias=False),
            nn.BatchNorm1d(width),
            nn.ReLU(),
            nn.MaxPool1d(3, 2, 1),
        )
        layers, ch = [], width
        for i in range(depth):
            out = width * min(2**i, 4)
            layers.append(_Block(ch, out, 1 if i == 0 else 2))
            ch = out
        self.layers = nn.Sequential(*layers)
        self.head = nn.Linear(ch, num_classes)
        self.embed_dim = ch

    def features(self, x: Tensor) -> Tensor:
        return self.layers(self.stem(x)).mean(-1)

    def forward(self, x: Tensor) -> Tensor:
        return self.head(self.features(x))

    Args:

    Returns:

    noise_power = noise_estimate ** 2

    snr = 10 * np.log10(signal_power / (noise_power + 1e-8))

    return float(snr)

    Estimate mode coverage using clustering.

    samples_flat = samples.reshape(len(samples), -1)

    def frechet_distance(mu1: np.ndarray, sigma1: np.ndarray, 
                    mu2: np.ndarray, sigma2: np.ndarray, eps: float = 1e-6) -> float:
    """
    Calculate Fréchet Distance between two Gaussian distributions.

    Args:
        mu1: Mean of first distribution
        sigma1: Covariance of first distribution
        mu2: Mean of second distribution
        sigma2: Covariance of second distribution
        eps: Small value for numerical stability

    Returns:
        Fréchet distance
    """
    mu1 = np.atleast_1d(mu1)
    mu2 = np.atleast_1d(mu2)

    sigma1 = np.atleast_2d(sigma1)
    sigma2 = np.atleast_2d(sigma2)

    diff = mu1 - mu2

    # Product might be almost singular
    covmean, _ = linalg.sqrtm(sigma1.dot(sigma2), disp=False)
    if not np.isfinite(covmean).all():
        offset = np.eye(sigma1.shape[0]) * eps
        covmean = linalg.sqrtm((sigma1 + offset).dot(sigma2 + offset))

    # Numerical error might give slight imaginary component
    if np.iscomplexobj(covmean):
        if not np.allclose(np.diagonal(covmean).imag, 0, atol=1e-3):
            m = np.max(np.abs(covmean.imag))
            raise ValueError(f"Imaginary component {m}")
        covmean = covmean.real

    tr_covmean = np.trace(covmean)

    return float(diff.dot(diff) + np.trace(sigma1) + np.trace(sigma2) - 2 * tr_covmean)

    def __init__(self, model: Any, device: str = "cuda"):
        self.model = model
        self.device = device
        self.results = {}

    def measure_inference_time(self, input_shape: tuple, n_runs: int = 100, warmup: int = 10):
        """
        Measure inference time.

        Args:
            input_shape: Input tensor shape
            n_runs: Number of runs for averaging
            warmup: Number of warmup runs
        """
        self.model.eval()
        dummy_input = torch.randn(input_shape).to(self.device)

        # Warmup
        with torch.no_grad():
            for _ in range(warmup):
                _ = self.model(dummy_input)

        # Measure
        if self.device == "cuda":
            torch.cuda.synchronize()

        start_time = time.time()
        with torch.no_grad():
            for _ in range(n_runs):
                _ = self.model(dummy_input)

        if self.device == "cuda":
            torch.cuda.synchronize()

        end_time = time.time()
        avg_time = (end_time - start_time) / n_runs

        self.results["inference_time_ms"] = avg_time * 1000
        self.results["throughput_samples_per_sec"] = input_shape[0] / avg_time

    def measure_memory_usage(self, input_shape: tuple):
        """Measure GPU memory usage."""
        if self.device != "cuda":
            return

        torch.cuda.reset_peak_memory_stats()
        dummy_input = torch.randn(input_shape).to(self.device)

        with torch.no_grad():
            _ = self.model(dummy_input)

        memory_allocated = torch.cuda.max_memory_allocated() / 1024**2  # MB
        memory_reserved = torch.cuda.max_memory_reserved() / 1024**2  # MB

        self.results["memory_allocated_mb"] = memory_allocated
        self.results["memory_reserved_mb"] = memory_reserved

    def count_parameters(self):
        """Count model parameters."""
        total_params = sum(p.numel() for p in self.model.parameters())
        trainable_params = sum(p.numel() for p in self.model.parameters() if p.requires_grad)

        self.results["total_parameters"] = total_params
        self.results["trainable_parameters"] = trainable_params
        self.results["total_parameters_millions"] = total_params / 1e6

        """Run full benchmark suite."""

    def plot_evaluation_results(
    metrics: dict,
    save_path: str | Path = None,
    figsize: tuple = (12, 8),
) -> plt.Figure:
    """
    Plot evaluation results as bar charts.

    Args:
        metrics: Dictionary of metric names and values
        save_path: Optional path to save figure
        figsize: Figure size

    Returns:
        matplotlib Figure object
    """
    fig, ax = plt.subplots(figsize=figsize)

    metric_names = list(metrics.keys())
    metric_values = list(metrics.values())

    bars = ax.bar(range(len(metric_names)), metric_values, alpha=0.7, edgecolor='black')

    # Color bars based on value (assuming lower is better for most metrics)
    colors = plt.cm.RdYlGn_r(np.linspace(0.2, 0.8, len(bars)))
    for bar, color in zip(bars, colors):
        bar.set_color(color)

    ax.set_xticks(range(len(metric_names)))
    ax.set_xticklabels(metric_names, rotation=45, ha='right')
    ax.set_ylabel("Metric Value")
    ax.set_title("Evaluation Metrics")
    ax.grid(True, alpha=0.3, axis='y')

    # Add value labels on bars
    for i, (bar, value) in enumerate(zip(bars, metric_values)):
        height = bar.get_height()
        ax.text(bar.get_x() + bar.get_width()/2., height,
               f'{value:.4f}',
               ha='center', va='bottom', fontsize=9)

    plt.tight_layout()

    if save_path:
        fig.savefig(save_path, dpi=300, bbox_inches='tight')

    return fig

    def plot_comparison_matrix(
    models: list[str],
    metrics: dict[str, list[float]],
    save_path: str | Path = None,
    figsize: tuple = (12, 8),
) -> plt.Figure:
    """
    Plot comparison matrix of multiple models across metrics.

    Args:
        models: List of model names
        metrics: Dictionary of metric names to lists of values (one per model)
        save_path: Optional path to save figure
        figsize: Figure size

    Returns:
        matplotlib Figure object
    """
    metric_names = list(metrics.keys())
    n_models = len(models)
    n_metrics = len(metric_names)

    # Create matrix
    matrix = np.array([metrics[m] for m in metric_names])

    fig, ax = plt.subplots(figsize=figsize)

    im = ax.imshow(matrix, cmap='RdYlGn_r', aspect='auto')

    # Set ticks
    ax.set_xticks(range(n_models))
    ax.set_yticks(range(n_metrics))
    ax.set_xticklabels(models, rotation=45, ha='right')
    ax.set_yticklabels(metric_names)

    # Add colorbar
    cbar = plt.colorbar(im, ax=ax)
    cbar.set_label("Metric Value", rotation=270, labelpad=20)

    # Add text annotations
    for i in range(n_metrics):
        for j in range(n_models):
            text = ax.text(j, i, f'{matrix[i, j]:.3f}',
                          ha="center", va="center", color="black", fontsize=9)

    ax.set_title("Model Comparison Matrix")
    plt.tight_layout()

    if save_path:
        fig.savefig(save_path, dpi=300, bbox_inches='tight')

    return fig
